#Snakefile

import yaml
import os

# Load config to access parameters
configfile: "config.yaml"

# Define output image filename
image_output = f"/stat3d/z{config['Z_SLICE_A']}_and_{config['Z_SLICE_B']}_level{config['LEVEL']}_morphology.tif"

# Define output CSV path and image output path 
csv_output = "/stat3d/QuPath_measurements.csv"

rule all:
    input:
        image_output,
        csv_output,
        "/stat3d/average_min_distance.csv",
        "/stat3d/average_nuclear_expansion.csv",
        "/stat3d/nuclei_diameter.csv",
		"/stat3d/z4_and_5_level2_morphology.seg.npy"

rule process_tiff:
    input:
        tiff = config["INPUT_TIFF"]
    output:
        image_output
    run:
        import subprocess
        subprocess.run([
            "python", "process_tiff.py",
            input.tiff,
            str(config["Z_SLICE_A"]),
            str(config["Z_SLICE_B"]),
            str(config["LEVEL"]),
            output[0]
        ])

rule generate_groovy_script:
    output:
        "QuPath_script.groovy"
    run:
        with open(output[0], "w") as f:
            f.write(f"""// QuPath_script.groovy
setImageType('FLUORESCENCE')
createFullImageAnnotation(true)
runPlugin('qupath.imagej.detect.cells.WatershedCellDetection', '{{"detectionImage":"Channel 1","backgroundByReconstruction":true,"backgroundRadius":{config["BACKGROUND_RADIUS"]},"medianRadius":{config["MEDIAN_RADIUS"]},"sigma":{config["SIGMA"]},"minArea":{config["MIN_AREA"]},"maxArea":{config["MAX_AREA"]},"threshold":{config["THRESHOLD"]},"watershedPostProcess":true,"cellExpansion":{config["CELL_EXPANSION"]},"includeNuclei":true,"smoothBoundaries":true,"makeMeasurements":true}}')
saveDetectionMeasurements('{csv_output}')
""")

rule run_qupath_analysis:
    input:
        image=image_output,
        script="QuPath_script.groovy"
    output:
        csv_output
    shell:
        "QuPath script --image {input.image} {input.script}"

rule image_measurements:
    input:
        csv=csv_output
    output:
        min_distance="/stat3d/average_min_distance.csv",
        nuclear_expansion="/stat3d/average_nuclear_expansion.csv",
        nuclei_diameter="/stat3d/nuclei_diameter.csv"
    shell:
        "Rscript image_measurements.R {input.csv} {output.min_distance} {output.nuclear_expansion} {output.nuclei_diameter}"

rule run_cellpose:
    input:
        images_dir = config["directory"]
    output:
        seg_npy = os.path.join(config["directory"], config["cellpose_img_filter"] + ".seg.npy")
    params:
        diameter = config["cellpose_diameter"],
        img_filter = config["cellpose_img_filter"],
        use_gpu = "--use_gpu" if config["cellpose_use_gpu"] else ""
    shell:
        """
        python -m cellpose \
            --dir {input.images_dir} \
            --pretrained_model nuclei \
            --chan 0 \
            --chan2 0 \
            --img_filter {params.img_filter} \
            --diameter {params.diameter} \
            --do_3D \
            --save_tif \
            --verbose \
            {params.use_gpu}
        """
